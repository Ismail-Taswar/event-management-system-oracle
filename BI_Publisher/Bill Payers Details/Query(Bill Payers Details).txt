SELECT        Booking_m_id, Client_name, Event_name, Venue_name, event_start_date,
              event_end_date,  (nvl(DUE, 0)) Booking_Due  , ((select   nvl(sum(amount), 0)  
              from    receive  where    booking_m_id =a.booking_m_id))Installment, 
              (nvl(DUE, 0)  - (select   nvl(sum(amount), 0)  
              from    receive  where    booking_m_id =a.booking_m_id)) Receiveable_Bill            
FROM          BOOKING_LOV_VIEW a
where         to_char(EVENT_END_DATE, 'MM/DD/YYYY') <= to_char(sysdate,  'MM/DD/YYYY') 
and           BOOKING_M_ID NOT IN ( SELECT     BOOKING_M_ID
                                   FROM            BILL  )  
order by     BOOKING_M_ID
