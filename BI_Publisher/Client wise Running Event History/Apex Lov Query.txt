select   distinct client_name || ' -( '|| (select    count(booking_m_id)
                                            from      booking_m bm, client cl 
                                            where     bm.client_id = cl.client_id 
                                            and       client_name  =  bkc.client_name 
                                            and       TO_CHAR(Event_start_date, 'MM/DD/RRRR') <= TO_CHAR(SYSDATE, 'MM/DD/RRRR')
                                            and       TO_CHAR(EVENT_END_DATE, 'MM/DD/RRRR') >= TO_CHAR(SYSDATE, 'MM/DD/RRRR')) ||' )'Total_Booking, Client_name
from       biller_details_view BKC
where      TO_CHAR(Event_start_date, 'MM/DD/RRRR') <= TO_CHAR(SYSDATE, 'MM/DD/RRRR')
and        TO_CHAR(EVENT_END_DATE, 'MM/DD/RRRR') >= TO_CHAR(SYSDATE, 'MM/DD/RRRR')
